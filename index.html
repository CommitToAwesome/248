2. Chaos Proxy: Zero-Copy Packet Mutation

The proxy uses FlatBuffers to achieve zero-copy mutation. Deserializing and reserializing Protobufs would cause Garbage Collection spikes, ruining latency. Instead, the proxy calculates memory offsets, mutates coordinates directly in the byte array, and recalculates the CRC32 checksum.

Go Implementation: FlatBuffer Intercept

func spoofPacket(rawPacket []byte, spoofOffsetX float32) []byte {
    packetLen := len(rawPacket)
    
    // 1. Strip the 4-byte checksum from the tail
    payload := rawPacket[:packetLen-4]

    // 2. Point to FlatBuffer root (Zero Memory Allocation)
    unit := sim.GetRootAsUnitUpdate(payload, 0)

    // 3. Zero-Copy Mutation: Overwrite bytes directly
    if unit.IsHostile() {
        currentX := unit.X()
        unit.MutateX(currentX + spoofOffsetX)
    }

    // 4. Recalculate Engine Checksum to prevent drop
    newChecksum := crc32.ChecksumIEEE(payload)
    rawPacket[packetLen-4] = byte(newChecksum >> 24)
    rawPacket[packetLen-3] = byte(newChecksum >> 16)
    rawPacket[packetLen-2] = byte(newChecksum >> 8)
    rawPacket[packetLen-1] = byte(newChecksum)

    return rawPacket
}
